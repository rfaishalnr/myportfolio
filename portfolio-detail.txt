  <script>
    // Fungsi untuk membuka zoom
    function openZoom(imgSrc, caption) {
      const modal = document.getElementById('zoomModal');
      const modalImg = document.getElementById('zoomImage');
      const captionText = document.getElementById('captionText');
      
      modal.style.display = "block";
      modalImg.src = imgSrc;
      captionText.innerHTML = caption || '';
      
      // Tambahkan event listener untuk menutup modal ketika klik di luar gambar
      modal.onclick = function(event) {
        if (event.target === modal) {
          closeZoom();
        }
      };
    }
    
    // Fungsi untuk menutup zoom
    function closeZoom() {
      document.getElementById('zoomModal').style.display = "none";
    }
    
    // Tambahkan listener untuk escape key
    document.addEventListener('keydown', function(event) {
      if (event.key === "Escape") {
        closeZoom();
      }
    });
    
    // Animasi elemen saat scroll
    document.addEventListener('DOMContentLoaded', function() {
      const fadeElements = document.querySelectorAll('.fade-in');
      
      // Animasi elemen pertama kali load
      fadeElements.forEach(function(element) {
        element.style.opacity = '0';
        setTimeout(function() {
          element.style.opacity = '1';
        }, parseInt(element.style.animationDelay || '0') * 1000);
      });
    });
  </script>
